\newglossaryentry{gls:sortition-legislature}{name={Sortition Legislature}, description={The national legislative body, made up of six hundred members selected by sortition and serving staggered three-year terms. Established by Article \ref{art:sortition-legislature}}}

\newglossaryentry{gls:constitutional-court}{name={Constitutional Court}, description={The twenty-one-justice body reviewing legislation for constitutional compliance. Established by Article \ref{art:const-court}}}

\newglossaryentry{gls:constitutional-assembly}{name={Constitutional Assembly}, description={The thousand-person body convened every twenty-five years to review and revise foundational principles. Established by Article \ref{art:const-assembly}}}

\newglossaryentry{gls:electoral-commons}{name={Electoral Commons}, description={The Commons institution responsible for maintaining the civic membership roll and conducting all sortition. Established by Article \ref{art:sortition-design}, \ref{art:sortition-design:s-electoral-commons}}}

\newglossaryentry{gls:qualification-commons}{name={Qualification Commons}, description={The permanent Commons institution maintaining qualification pools for sortition bodies requiring demonstrated competence. Established by Article \ref{art:sortition-design}, \ref{art:sortition-design:s-qualification-commons}}}

\newglossaryentry{gls:rubric-body}{name={Rubric Body}, description={The nine-member body that designs and revises qualification rubrics for the Qualification Commons. Established by Article \ref{art:sortition-design}, \ref{art:sortition-design:s-qualification-commons}}}

\newglossaryentry{gls:citizen-oversight-panels}{name={Citizen Oversight Panel}, description={Twelve-member panels embedded within every governance body and Implementation Council. Established by Article \ref{art:citizen-oversight-panels}}}

\newglossaryentry{gls:cross-panel-coordination-body}{name={Cross-Panel Coordination Body}, description={A body of rotating representatives from each Citizen Oversight Panel, sharing observed patterns across governance bodies. Established by Article \ref{art:citizen-oversight-panels}, \ref{art:citizen-oversight-panels:s-cross-panel-coordination}}}

\newglossaryentry{gls:transparency-engine}{name={Transparency Engine}, description={The open-source, cryptographically tamper-evident system through which all exercises of power are logged and disclosed in real time. Established by Article \ref{art:transparency-engine}}}

\newglossaryentry{gls:transparency-engine-oversight-body}{name={Transparency Engine Oversight Body}, description={The nine-member sortition body that may add logging categories to the Transparency Engine. Established by Article \ref{art:transparency-engine}, \ref{art:transparency-engine:s-significance-thresholds-logging}}}

\newglossaryentry{gls:whistleblower-infrastructure}{name={Whistleblower Infrastructure}, description={The constitutionally protected anonymous reporting channel embedded within the Transparency Engine. Established by Article \ref{art:transparency-engine}, \ref{art:transparency-engine:s-whistleblower-infrastructure}}}

\newglossaryentry{gls:whistleblower-infrastructure-panel}{name={Whistleblower Infrastructure Panel}, description={The five-member body governing the Whistleblower Infrastructure. Established by Article \ref{art:transparency-engine}, \ref{art:transparency-engine:s-whistleblower-infrastructure}}}

\newglossaryentry{gls:civic-initiative-infrastructure}{name={Civic Initiative Infrastructure}, description={The Commons institution providing verified channels for the civic membership to trigger constitutional interventions. Established by Article \ref{art:civic-initiative-infrastructure}}}

\newglossaryentry{gls:civic-initiative-oversight-body}{name={Civic Initiative Oversight Body}, description={The nine-member body governing the Civic Initiative Infrastructure. Established by Article \ref{art:civic-initiative-infrastructure}, \ref{art:civic-initiative-infrastructure:s-governance}}}

\newglossaryentry{gls:emergency-review-panel}{name={Emergency Review Panel}, description={A one-hundred-fifty-person panel reviewing actions taken under emergency authorization. Established by Article \ref{art:civic-initiative-infrastructure}, \ref{art:civic-initiative-infrastructure:s-constitutional-intervention-types}}}

\newglossaryentry{gls:minority-impact-assessment-body}{name={Minority Impact Assessment Body}, description={The twenty-one-member body assessing whether local decisions significantly harm an identifiable minority. Established by Article \ref{art:local-democracy}, \ref{art:local-democracy:s-minority-impact-assessment}}}

\newglossaryentry{gls:community-justice-bodies}{name={Community Justice Bodies}, description={Locally selected bodies facilitating restorative justice processes within local democratic governance. Established by Article \ref{art:local-democracy}, \ref{art:local-democracy:s-community-justice-bodies}}}

\newglossaryentry{gls:exemption-review-panel}{name={Exemption Review Panel}, description={The nine-member panel within the Electoral Commons reviewing disputed hardship exemption certifications. Established by Article \ref{art:sortition-design}, \ref{art:sortition-design:s-mandatory-service-civic}}}

\newglossaryentry{gls:expert-advisory-corps}{name={Expert Advisory Corps}, description={The permanent body of domain specialists advising the Sortition Legislature. Established by Article \ref{art:sortition-legislature}, \ref{art:sortition-legislature:s-expert-advisory-corps}}}

\newglossaryentry{gls:meta-review-chamber}{name={Meta-Review Chamber}, description={The thirty-member body reviewing Constitutional Court decisions for quality of reasoning and consistency. Established by Article \ref{art:const-court}, \ref{art:const-court:s-meta-review-chamber}}}

\newglossaryentry{gls:implementation-councils}{name={Implementation Council}, description={Domain-specific bodies translating legislative decisions into operational reality. Established by Article \ref{art:implementation-councils}}}

\newglossaryentry{gls:coordination-assembly}{name={Coordination Assembly}, description={The body of rotating Implementation Council representatives managing cross-domain issues. Established by Article \ref{art:implementation-councils}, \ref{art:implementation-councils:s-cross-domain-coordination}}}

\newglossaryentry{gls:emergency-threshold-verification-body}{name={Emergency Threshold Verification Body}, description={The sixteen-expert body that verifies objective triggering conditions for emergency protocols. Established by Article \ref{art:emergency-protocols}, \ref{art:emergency-protocols:s-automatic-activation}}}

\newglossaryentry{gls:residual-coordination-council}{name={Residual Coordination Council}, description={The five-person body with narrowly limited fallback coordination authority in novel cross-domain emergencies. Established by Article \ref{art:residual-coordination}}}

\newglossaryentry{gls:cooperative-development-bank}{name={Cooperative Development Bank}, description={The publicly chartered bank providing at-cost startup financing and working capital to cooperatives. Established by Article \ref{art:worker-coop-sector}, \ref{art:worker-coop-sector:s-cooperative-development-bank}}}

\newglossaryentry{gls:national-cooperative-standards-body}{name={National Cooperative Standards Body}, description={The body setting and enforcing the minimum size threshold at which an enterprise must organize as a worker cooperative. Established by Article \ref{art:regulatory-bodies}, \ref{art:regulatory-bodies:s-major-regulatory-domains}}}

\newglossaryentry{gls:housing-implementation-council}{name={Housing Implementation Council}, description={The Implementation Council responsible for housing, tenure assessment, and Commons-Allocated Housing. Established by Article \ref{art:open-presence}, \ref{art:open-presence:s-commons-allocated-housing}}}

\newglossaryentry{gls:housing-appeals-panel}{name={Housing Appeals Panel}, description={The nine-member body hearing appeals of Commons-Allocated Housing denials. Established by Article \ref{art:open-presence}, \ref{art:open-presence:s-commons-allocated-housing}}}

\newglossaryentry{gls:justice-standards-council}{name={Justice Standards Council}, description={The body setting minimum dignity standards for transitional housing and restorative justice processes. Established by Article \ref{art:justice-restoration}, \ref{art:justice-restoration:s-justice-standards-council}}}

\newglossaryentry{gls:public-banking-system}{name={Public Banking System}, description={The Commons institution providing savings, checking, and personal lending services at cost. Established by Article \ref{art:finance-money}, \ref{art:finance-money:s-public-banking}}}

\newglossaryentry{gls:monetary-authority}{name={Monetary Authority}, description={The democratically governed public body responsible for money creation. Established by Article \ref{art:finance-money}, \ref{art:finance-money:s-monetary-authority}}}

\newglossaryentry{gls:commons-investment-fund}{name={Commons Investment Fund}, description={The fund financing long-term infrastructure investment. Established by Article \ref{art:finance-money}, \ref{art:finance-money:s-commons-investment-fund}}}

\newglossaryentry{gls:financial-regulatory-body}{name={Financial Regulatory Body}, description={The enforcement authority for the prohibitions on rent-extractive financial activity, and for cooperative market dominance. Established by Article \ref{art:finance-money}, \ref{art:finance-money:s-enforcement}}}

\newglossaryentry{gls:environmental-regulatory-body}{name={Environmental Regulatory Body}, description={The body setting and enforcing emissions, water quality, land use, and extraction standards. Established by Article \ref{art:regulatory-bodies}, \ref{art:regulatory-bodies:s-major-regulatory-domains}}}

\newglossaryentry{gls:public-health-safety-regulatory-body}{name={Public Health and Safety Regulatory Body}, description={The body overseeing pharmaceutical, medical device, food, and public health safety. Established by Article \ref{art:regulatory-bodies}, \ref{art:regulatory-bodies:s-major-regulatory-domains}}}

\newglossaryentry{gls:worker-safety-regulatory-body}{name={Worker Safety Regulatory Body}, description={The body setting minimum workplace safety floors. Established by Article \ref{art:regulatory-bodies}, \ref{art:regulatory-bodies:s-major-regulatory-domains}}}

\newglossaryentry{gls:foreign-relations-council}{name={Foreign Relations Council}, description={The twelve-member body managing diplomatic relationships and negotiating international agreements. Established by Article \ref{art:foreign-relations}}}

\newglossaryentry{gls:treaty-review-body}{name={Treaty Review Body}, description={The twenty-member body reviewing proposed international agreements for compliance with the anti-imperialism principle. Established by Article \ref{art:anti-imperialism}, \ref{art:anti-imperialism:s-treaty-review}}}

\newglossaryentry{gls:nuclear-use-authorization-council}{name={Nuclear Use Authorization Council}, description={The eight-member body holding sole nuclear launch authorization. Established by Article \ref{art:defense}, \ref{art:defense:s-nuclear-use-authorization}}}

\newglossaryentry{gls:defensive-posture-review-body}{name={Defensive Posture Review Body}, description={The sixteen-member body conducting mandatory public review of military force structure and capability. Established by Article \ref{art:defense}, \ref{art:defense:s-defensive-posture-review}}}

\newglossaryentry{gls:security-investigation-governing-board}{name={Security Investigation Governing Board}, description={The nine-member body that sets investigation priorities and authorizes opening and closing of security investigations. Established by Article \ref{art:internal-security}, \ref{art:internal-security:s-investigation-body}}}

\newglossaryentry{gls:security-enforcement-review-panel}{name={Security Enforcement Review Panel}, description={The nine-member body that independently authorizes or declines enforcement actions on investigation findings. Established by Article \ref{art:internal-security}, \ref{art:internal-security:s-security-enforcement-review}}}

\newglossaryentry{gls:land-back-negotiation-office}{name={Land Back Negotiation Office}, description={The office jointly structured by the Commonwealth and indigenous nations to negotiate the return of territorial claims. Established by Article \ref{art:indigenous-sov}}}

\newglossaryentry{gls:reparations-design-council}{name={Reparations Design Council}, description={The affected-community-majority body designing and overseeing reparative resource allocation. Established by Article \ref{art:reparations}}}
